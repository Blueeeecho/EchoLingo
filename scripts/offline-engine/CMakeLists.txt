cmake_minimum_required(VERSION 3.20)
project(echolingo_offline_engine LANGUAGES C CXX)

if(NOT DEFINED ECHOLINGO_MOZILLA_SOURCE)
  message(FATAL_ERROR "Set ECHOLINGO_MOZILLA_SOURCE to the pinned translations source directory")
endif()
if(NOT DEFINED ECHOLINGO_PCRE2_SOURCE)
  message(FATAL_ERROR "Set ECHOLINGO_PCRE2_SOURCE to the pinned PCRE2 source directory")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(WIN32)
  # Keep the generic code below AVX2; intgemm/ruy select supported ISA kernels.
  set(BUILD_ARCH core2 CACHE STRING "Windows x64 baseline" FORCE)
  add_compile_definitions(NOMINMAX)
endif()
set(GIT_SUBMODULE OFF CACHE BOOL "Use only hash-verified prepared source" FORCE)
set(COMPILE_WASM OFF CACHE BOOL "Native process" FORCE)
set(USE_WASM_COMPATIBLE_SOURCE ON CACHE BOOL "Minimal inference source" FORCE)
set(COMPILE_WITHOUT_EXCEPTIONS OFF CACHE BOOL "Allow protocol errors" FORCE)
set(COMPILE_UNIT_TESTS OFF CACHE BOOL "No upstream test programs" FORCE)
set(COMPILE_TESTS OFF CACHE BOOL "No upstream test programs" FORCE)
set(COMPILE_SERVER OFF CACHE BOOL "No network server" FORCE)
set(COMPILE_EXAMPLES OFF CACHE BOOL "No examples" FORCE)
set(COMPILE_CUDA OFF CACHE BOOL "CPU only" FORCE)
set(USE_MKL OFF CACHE BOOL "No proprietary BLAS dependency" FORCE)
set(USE_ONNX_SGEMM OFF CACHE BOOL "Use native float GEMM" FORCE)
set(USE_DOXYGEN OFF CACHE BOOL "No documentation build" FORCE)
set(USE_STATIC_LIBS ON CACHE BOOL "Self-contained runtime" FORCE)
set(SPM_BUILD_LIBRARY_ONLY ON CACHE BOOL "Tokenizer library only" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "No optional allocator" FORCE)
set(RUY_MINIMAL_BUILD ON CACHE BOOL "Only inference library" FORCE)
set(RUY_FIND_CPUINFO OFF CACHE BOOL "Use pinned vendored cpuinfo" FORCE)
set(RUY_PROFILER OFF CACHE BOOL "No profiling" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static dependencies" FORCE)
set(PCRE2_BUILD_PCRE2GREP OFF CACHE BOOL "No regex utility" FORCE)
set(PCRE2_BUILD_TESTS OFF CACHE BOOL "No regex tests" FORCE)
set(PCRE2_SUPPORT_JIT OFF CACHE BOOL "No executable memory" FORCE)
set(PCRE2_STATIC_RUNTIME ON CACHE BOOL "Static Windows runtime" FORCE)
set(PCRE2_SUPPORT_LIBZ OFF CACHE BOOL "No compression dependency" FORCE)
set(PCRE2_SUPPORT_LIBBZ2 OFF CACHE BOOL "No compression dependency" FORCE)
set(PCRE2_SUPPORT_LIBREADLINE OFF CACHE BOOL "No interactive shell dependency" FORCE)
add_subdirectory("${ECHOLINGO_PCRE2_SOURCE}" "pcre2" EXCLUDE_FROM_ALL)
set(SSPLIT_USE_INTERNAL_PCRE2 OFF CACHE BOOL "Use separately pinned PCRE2" FORCE)
set(PCRE2_LIBRARIES pcre2-8-static CACHE STRING "Pinned PCRE2 target" FORCE)
set(PCRE2_INCLUDE_DIRS "${CMAKE_CURRENT_BINARY_DIR}/pcre2" CACHE PATH "Pinned PCRE2 generated headers" FORCE)
if(NOT APPLE)
  set(USE_RUY ON CACHE BOOL "Portable float GEMM" FORCE)
  set(USE_RUY_SGEMM ON CACHE BOOL "No external BLAS required" FORCE)
endif()
add_subdirectory("${ECHOLINGO_MOZILLA_SOURCE}/inference" "mozilla" EXCLUDE_FROM_ALL)
# Upstream's native architecture definitions are directory scoped to marian.
# Its public headers are also compiled by Bergamot and this CLI.
target_compile_definitions(bergamot-translator-source PUBLIC COMPILE_CPU=1 USE_SENTENCEPIECE _USE_INTERNAL_STRING_VIEW)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm|arm64|aarch64|ARM64)")
  target_compile_definitions(bergamot-translator-source PUBLIC ARM FMA SSE)
else()
  target_compile_definitions(bergamot-translator-source PUBLIC USE_INTGEMM=1)
endif()
add_executable(echolingo-offline-engine main.cpp)
target_include_directories(echolingo-offline-engine PRIVATE "${ECHOLINGO_JSON_SOURCE}")
target_link_libraries(echolingo-offline-engine PRIVATE bergamot-translator-source)
if(MSVC)
  target_compile_options(echolingo-offline-engine PRIVATE /utf-8 /EHsc)
endif()
